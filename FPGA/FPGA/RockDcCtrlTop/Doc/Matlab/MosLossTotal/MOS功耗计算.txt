% 1>定义电路参数：如输入电压、输出电压、输出电流、开关频率、占空比、环境温度等。
% 2>定义MOS管参数：如Rds(on)（在特定结温和Vgs下）、开关时间（tr, tf）或开关能量（Eon, Eoff）、栅极电荷Qg、热阻等。
% 3>计算导通损耗：P_cond = Irms^2 * Rds(on) * D
% 4>计算开关损耗：P_sw = (Eon + Eoff) * f_sw 或者使用近似公式 P_sw = 0.5 * Vds * Id * (tr+tf) * f_sw
% 5>计算驱动损耗：P_drive = Qg * Vgs * f_sw
% 6>计算总损耗：P_total = P_cond + P_sw + P_drive
% 7>计算结温：Tj = Ta + P_total * Rth_ja（其中Rth_ja是结到环境的热阻）

%% 辅助函数：扫描开关频率对损耗的影响
function frequency_sweep()
% 分析开关频率对损耗的影响
    fprintf('\n=== 开关频率扫描分析 ===\n');
    
    % 基础配置
    base_config.Vin = 24;
    base_config.Vout = 12;
    base_config.I_load = 5;
    base_config.Vgs = 10;
    base_config.Ta = 40;
    
    % MOSFET参数 (Infineon IPP320N20N3)
    base_config.Rds_on = 0.0032;    % 3.2mΩ
    base_config.Qg = 110;           % 110nC
    base_config.Qgs = 35;           % 35nC
    base_config.Qgd = 52;           % 52nC
    base_config.tr = 25e-9;         % 25ns
    base_config.tf = 20e-9;         % 20ns
    base_config.Coss = 680e-12;     % 680pF
    base_config.Rth_ja = 40;        % 40°C/W
    base_config.plot_results = false;
    
    % 频率扫描范围
    fsw_range = [50e3, 100e3, 200e3, 300e3, 500e3, 1e6];
    
    % 存储结果
    results = zeros(length(fsw_range), 5);
    
    fprintf('频率(kHz)\t导通损耗\t开关损耗\t总损耗\t结温(°C)\n');
    fprintf('---------------------------------------------------\n');
    
    for i = 1:length(fsw_range)
        config = base_config;
        config.fsw = fsw_range(i);
        config.Tj_est = 80;
        
        [total_loss, breakdown] = calculate_mosfet_loss(config);
        
        results(i, :) = [fsw_range(i)/1e3, breakdown.cond_loss, ...
                        breakdown.sw_loss, total_loss, breakdown.junction_temp];
        
        fprintf('%.0f\t\t%.3f\t\t%.3f\t\t%.3f\t%.1f\n', results(i, :));
    end
    
    % 绘制频率扫描图
    figure;
    subplot(2,1,1);
    plot(results(:,1), results(:,2), 'b-o', 'LineWidth', 2, 'DisplayName', '导通损耗');
    hold on;
    plot(results(:,1), results(:,3), 'r-s', 'LineWidth', 2, 'DisplayName', '开关损耗');
    plot(results(:,1), results(:,4), 'k-^', 'LineWidth', 2, 'DisplayName', '总损耗');
    xlabel('开关频率 (kHz)');
    ylabel('损耗 (W)');
    title('开关频率对损耗的影响');
    legend;
    grid on;
    
    subplot(2,1,2);
    plot(results(:,1), results(:,5), 'g-d', 'LineWidth', 2);
    xlabel('开关频率 (kHz)');
    ylabel('结温 (°C)');
    title('开关频率对结温的影响');
    grid on;
end   



%% 辅助函数：扫描负载电流对效率的影响
function efficiency_vs_load(base_config)
    fprintf('\n=== 负载电流扫描分析 ===\n');
    
    % 基础配置
    base_config.Vin = 12;
    base_config.Vout = 24;
    base_config.fsw = 100e3;
    base_config.L = 10e-6;
    base_config.RL = 0.005;
    base_config.plot_results = false;
    
    % MOSFET参数
    base_config.Rds_on_ctrl = 0.005;
    base_config.Rds_on_sync = 0.003;
    base_config.Qg_ctrl = 25;
    base_config.Qg_sync = 20;
    base_config.Qgd_ctrl = 12;
    base_config.Qgd_sync = 10;
    base_config.Coss_ctrl = 300e-12;
    base_config.Coss_sync = 400e-12;
    base_config.Qrr = 15;
    
    % 扫描负载电流
    I_out_range = [0.5, 1, 2, 3, 4, 5, 6, 7, 8];
    efficiencies = zeros(size(I_out_range));
    ctrl_losses = zeros(size(I_out_range));
    sync_losses = zeros(size(I_out_range));
    
    fprintf('负载电流(A)\t效率(%%)\t控制管损耗(W)\t同步管损耗(W)\n');
    fprintf('--------------------------------------------------------\n');
    
    for i = 1:length(I_out_range)
        config = base_config;
        config.I_out = I_out_range(i);
        
        [~, results] = calculate_sync_boost_loss(config);
        
        efficiencies(i) = results.efficiency;
        ctrl_losses(i) = results.control.total_loss;
        sync_losses(i) = results.sync.total_loss;
        
        fprintf('%.1f\t\t%.2f\t\t%.3f\t\t%.3f\n', ...
            I_out_range(i), efficiencies(i), ctrl_losses(i), sync_losses(i));
    end
    
    % 绘制效率曲线
    figure;
    subplot(2,1,1);
    plot(I_out_range, efficiencies, 'b-o', 'LineWidth', 2);
    xlabel('输出电流 (A)');
    ylabel('效率 (%)');
    title('同步Boost效率 vs 负载电流');
    grid on;
    
    subplot(2,1,2);
    plot(I_out_range, ctrl_losses, 'r-s', 'LineWidth', 2, 'DisplayName', '控制管损耗');
    hold on;
    plot(I_out_range, sync_losses, 'g-^', 'LineWidth', 2, 'DisplayName', '同步管损耗');
    xlabel('输出电流 (A)');
    ylabel('损耗 (W)');
    title('MOSFET损耗 vs 负载电流');
    legend;
    grid on;
end



%%

function [total_loss, loss_breakdown] = calculate_sync_boost_loss(varargin)
% 计算同步Boost变换器上下MOSFET的功率损耗
% 用法: [total_loss, loss_breakdown] = calculate_sync_boost_loss('param1', value1, ...)

    %% 参数解析
    if nargin == 1 && isstruct(varargin{1})
        config = varargin{1};
    else
        config = parse_sync_boost_parameters(varargin{:});
    end
    
    %% 从配置中提取参数
    Vin         = config.Vin        ;   % 输入电压 (V)
    Vout        = config.Vout       ;   % 输出电压 (V)
    I_out       = config.I_out      ;   % 输出电流 (A)
    fsw         = config.fsw        ;   % 开关频率 (Hz)
    L           = config.L          ;   % 电感值 (H)
    RL          = config.RL         ;   % 电感等效串联电阻 (Ω)
    
    % 控制MOSFET参数 (下管 - 主动开关)
    Rds_on_ctrl = config.Rds_on_ctrl;   % 控制管导通电阻 (Ω)
    Qg_ctrl     = config.Qg_ctrl    ;   % 控制管栅极电荷 (nC)
    Qgd_ctrl    = config.Qgd_ctrl   ;   % 控制管栅漏电荷 (nC)
    tr_ctrl     = config.tr_ctrl    ;   % 控制管上升时间 (s)
    tf_ctrl     = config.tf_ctrl    ;   % 控制管下降时间 (s)
    Coss_ctrl   = config.Coss_ctrl  ;   % 控制管输出电容 (F)
    
    % 同步MOSFET参数 (上管 - 同步整流)
    Rds_on_sync = config.Rds_on_sync;   % 同步管导通电阻 (Ω)
    Qg_sync     = config.Qg_sync    ;   % 同步管栅极电荷 (nC)
    Qgd_sync    = config.Qgd_sync   ;   % 同步管栅漏电荷 (nC)
    tr_sync     = config.tr_sync    ;   % 同步管上升时间 (s)
    tf_sync     = config.tf_sync    ;   % 同步管下降时间 (s)
    Coss_sync   = config.Coss_sync  ;   % 同步管输出电容 (F)
    
    % 通用参数
    Vgs         = config.Vgs        ;   % 栅极驱动电压 (V)
    Tj_est      = config.Tj_est     ;   % 估计结温 (°C)
    Ta          = config.Ta         ;   % 环境温度 (°C)
    Rth_ja      = config.Rth_ja     ;   % 结到环境热阻 (°C/W)
    
    %% 计算同步Boost基本工作参数
    D = 1 - (Vin / Vout);       % 占空比 (控制管导通占空比)
    I_in = I_out / (1 - D);     % 输入平均电流 (A)
    
    % 电感电流纹波
    delta_IL = (Vin * D) / (fsw * L);  % 电感电流纹波峰值 (A)
    IL_avg = I_in;                     % 电感平均电流 = 输入平均电流 (A)
    IL_rms = sqrt(IL_avg^2 + (delta_IL^2 / 12)); % 电感电流RMS值
    
    fprintf('\n=== 同步Boost工作参数 ===\n');
    fprintf('占空比 (控制管): %.3f\n', D);
    fprintf('输入平均电流: %.3f A\n', I_in);
    fprintf('电感平均电流: %.3f A\n', IL_avg);
    fprintf('电感电流纹波: ±%.3f A\n', delta_IL/2);
    fprintf('电感电流RMS: %.3f A\n', IL_rms);
    
    %% 控制MOSFET (下管) 损耗计算
    fprintf('\n=== 控制MOSFET (下管) 损耗计算 ===\n');
    
    % 1. 导通损耗--------------------------------------------------------------
    % 控制管只在(D)时间内导通，电流为电感电流
    Rds_on_ctrl_actual = Rds_on_ctrl * (1 + 0.007 * (Tj_est - 25));
    P_cond_ctrl = IL_rms^2 * Rds_on_ctrl_actual * D;
    fprintf('控制管导通损耗: %.3f W\n', P_cond_ctrl);
    
    % 2. 开关损耗---------------------------------------------------------------
    % 控制管开关时承受的电压为输出电压Vout
    V_sw_ctrl = Vout;
    
    % 使用栅极电荷法计算开关损耗
    V_plateau = 4; % 典型米勒平台电压 (V)
  E_on_ctrl = Qgd_ctrl * 1e-9 * V_plateau;        % 开通能量
  E_off_ctrl = Qgd_ctrl * 1e-9 * (V_sw_ctrl - V_plateau); % 关断能量
  P_sw_ctrl = (E_on_ctrl + E_off_ctrl) * fsw;
   % E_on_ctrl = 1/2 * V_sw_ctrl * IL_rms * tr_ctrl ;        % 开通能量
   % E_off_ctrl = 1/2 * V_sw_ctrl * IL_rms * tf_ctrl;  % 关断能量
   % P_sw_ctrl = (E_on_ctrl + E_off_ctrl) * fsw;
    fprintf('控制管开关损耗: %.3f W\n', P_sw_ctrl);
    
    % 3. 输出电容损耗
    P_coss_ctrl = 0.5 * Coss_ctrl * V_sw_ctrl^2 * fsw;
    fprintf('控制管Coss损耗: %.3f W\n', P_coss_ctrl);
    
    % 4. 驱动损耗
    P_drive_ctrl = Qg_ctrl * 1e-9 * Vgs * fsw;
    fprintf('控制管驱动损耗: %.3f W\n', P_drive_ctrl);
    
    P_total_ctrl = P_cond_ctrl + P_sw_ctrl + P_coss_ctrl + P_drive_ctrl;
    fprintf('控制管总损耗: %.3f W\n', P_total_ctrl);
    
    %% 同步MOSFET (上管) 损耗计算
    fprintf('\n=== 同步MOSFET (上管) 损耗计算 ===\n');
    
    % 1. 导通损耗
    % 同步管在(1-D)时间内导通，电流为电感电流
    Rds_on_sync_actual = Rds_on_sync * (1 + 0.007 * (Tj_est - 25));
    P_cond_sync = IL_rms^2 * Rds_on_sync_actual * (1 - D);
    fprintf('同步管导通损耗: %.3f W\n', P_cond_sync);
    
    % 2. 开关损耗
    % 同步管开关损耗通常较小，因为体二极管先导通
    % 但反向恢复会产生额外损耗
    
    % a) 体二极管反向恢复损耗
    Qrr = config.Qrr;           % 反向恢复电荷 (nC)
    if Qrr > 0
        P_rr_sync = 0.5 * Qrr * 1e-9 * Vout * fsw;
    else
        % 估算反向恢复损耗
        P_rr_sync = 0.25 * (Qgd_sync * 1e-9 * Vout * fsw); % 约为控制管的25%
    end
    
    % b) 常规开关损耗 (较小)
    V_sw_sync = Vout; % 同步管承受输出电压
    E_sw_sync = 0.25 * (Qgd_sync * 1e-9 * V_sw_sync); % 同步管开关能量较小
    P_sw_sync = E_sw_sync * fsw + P_rr_sync;
    fprintf('同步管开关损耗: %.3f W (含反向恢复)\n', P_sw_sync);
    
    % 3. 输出电容损耗
    P_coss_sync = 0.5 * Coss_sync * V_sw_sync^2 * fsw;
    fprintf('同步管Coss损耗: %.3f W\n', P_coss_sync);
    
    % 4. 驱动损耗
    P_drive_sync = Qg_sync * 1e-9 * Vgs * fsw;
    fprintf('同步管驱动损耗: %.3f W\n', P_drive_sync);
    
    P_total_sync = P_cond_sync + P_sw_sync + P_coss_sync + P_drive_sync;
    fprintf('同步管总损耗: %.3f W\n', P_total_sync);
    
    %% 电感损耗
    P_inductor = IL_rms^2 * RL;
    fprintf('\n电感铜损: %.3f W\n', P_inductor);
    
    %% 总损耗和效率计算
    P_total = P_total_ctrl + P_total_sync + P_inductor;
    P_out = Vout * I_out;
    P_in = P_out + P_total;
    efficiency = P_out / P_in * 100;
    
    fprintf('\n=== 总损耗和效率 ===\n');
    fprintf('输出功率: %.3f W\n', P_out);
    fprintf('总损耗: %.3f W\n', P_total);
    fprintf('输入功率: %.3f W\n', P_in);
    fprintf('计算效率: %.2f%%\n', efficiency);
    
    %% 热计算
    Tj_ctrl = Ta + P_total_ctrl * Rth_ja;
    Tj_sync = Ta + P_total_sync * Rth_ja;
    
    fprintf('\n=== 热分析 ===\n');
    fprintf('环境温度: %.1f°C\n', Ta);
    fprintf('控制管结温: %.1f°C\n', Tj_ctrl);
    fprintf('同步管结温: %.1f°C\n', Tj_sync);
    
    %% 返回结果
    total_loss = P_total;
    
    loss_breakdown.control = struct(...
        'cond_loss', P_cond_ctrl, ...
        'sw_loss', P_sw_ctrl, ...
        'coss_loss', P_coss_ctrl, ...
        'drive_loss', P_drive_ctrl, ...
        'total_loss', P_total_ctrl, ...
        'junction_temp', Tj_ctrl);
    
    loss_breakdown.sync = struct(...
        'cond_loss', P_cond_sync, ...
        'sw_loss', P_sw_sync, ...
        'coss_loss', P_coss_sync, ...
        'drive_loss', P_drive_sync, ...
        'total_loss', P_total_sync, ...
        'junction_temp', Tj_sync);
    
    loss_breakdown.inductor = P_inductor;
    loss_breakdown.efficiency = efficiency;
    loss_breakdown.operating_params = struct(...
        'duty_cycle', D, ...
        'I_in_avg', I_in, ...
        'IL_rms', IL_rms, ...
        'delta_IL', delta_IL);
    
    %% 绘制结果
    if config.plot_results
        plot_sync_boost_results(loss_breakdown, config);
    end
end

function config = parse_sync_boost_parameters(varargin)
% 解析同步Boost参数
    p = inputParser;
    
    % 基本工作参数
    addParameter(p, 'Vin', 12, @isnumeric);
    addParameter(p, 'Vout', 24, @isnumeric);
    addParameter(p, 'I_out', 5, @isnumeric);
    addParameter(p, 'fsw', 100e3, @isnumeric);
    addParameter(p, 'L', 10e-6, @isnumeric);
    addParameter(p, 'RL', 0.005, @isnumeric); % 电感DCR
    
    % 控制MOSFET参数 (下管)
    addParameter(p, 'Rds_on_ctrl', 0.005, @isnumeric);   % 5mΩ
    addParameter(p, 'Qg_ctrl', 25, @isnumeric);          % 25nC
    addParameter(p, 'Qgd_ctrl', 12, @isnumeric);         % 12nC
    addParameter(p, 'tr_ctrl', 20e-9, @isnumeric);       % 20ns
    addParameter(p, 'tf_ctrl', 15e-9, @isnumeric);       % 15ns
    addParameter(p, 'Coss_ctrl', 300e-12, @isnumeric);   % 300pF
    
    % 同步MOSFET参数 (上管)
    addParameter(p, 'Rds_on_sync', 0.003, @isnumeric);   % 3mΩ (通常更小)
    addParameter(p, 'Qg_sync', 20, @isnumeric);          % 20nC
    addParameter(p, 'Qgd_sync', 10, @isnumeric);         % 10nC
    addParameter(p, 'tr_sync', 15e-9, @isnumeric);       % 15ns
    addParameter(p, 'tf_sync', 12e-9, @isnumeric);       % 12ns
    addParameter(p, 'Coss_sync', 400e-12, @isnumeric);   % 400pF
    addParameter(p, 'Qrr', 15, @isnumeric);              % 反向恢复电荷
    
    % 通用参数
    addParameter(p, 'Vgs', 10, @isnumeric);
    addParameter(p, 'Tj_est', 80, @isnumeric);
    addParameter(p, 'Ta', 40, @isnumeric);
    addParameter(p, 'Rth_ja', 40, @isnumeric);
    addParameter(p, 'plot_results', true, @islogical);
    
    parse(p, varargin{:});
    config = p.Results;
end

function plot_sync_boost_results(loss_breakdown, config)
% 绘制同步Boost损耗分析结果
    figure('Position', [100, 100, 1200, 800]);
    
    % 子图1: 总损耗分布
    subplot(2,3,1);
    losses_total = [loss_breakdown.control.total_loss, ...
                   loss_breakdown.sync.total_loss, ...
                   loss_breakdown.inductor];
    labels_total = {'控制管', '同步管', '电感'};
    pie(losses_total, labels_total);
    title('总损耗分布');
    
    % 子图2: 控制管损耗细分
    subplot(2,3,2);
    losses_ctrl = [loss_breakdown.control.cond_loss, ...
                  loss_breakdown.control.sw_loss, ...
                  loss_breakdown.control.coss_loss, ...
                  loss_breakdown.control.drive_loss];
    labels_ctrl = {'导通', '开关', 'Coss', '驱动'};
    pie(losses_ctrl, labels_ctrl);
    title('控制管损耗细分');
    
    % 子图3: 同步管损耗细分
    subplot(2,3,3);
    losses_sync = [loss_breakdown.sync.cond_loss, ...
                  loss_breakdown.sync.sw_loss, ...
                  loss_breakdown.sync.coss_loss, ...
                  loss_breakdown.sync.drive_loss];
    labels_sync = {'导通', '开关', 'Coss', '驱动'};
    pie(losses_sync, labels_sync);
    title('同步管损耗细分');
    
    % 子图4: 损耗对比柱状图
    subplot(2,3,4);
    categories = {'导通损耗', '开关损耗', 'Coss损耗', '驱动损耗'};
    ctrl_data = [loss_breakdown.control.cond_loss, loss_breakdown.control.sw_loss, ...
                loss_breakdown.control.coss_loss, loss_breakdown.control.drive_loss];
    sync_data = [loss_breakdown.sync.cond_loss, loss_breakdown.sync.sw_loss, ...
                loss_breakdown.sync.coss_loss, loss_breakdown.sync.drive_loss];
    
    bar_data = [ctrl_data; sync_data]';
    bar(bar_data);
    set(gca, 'XTickLabel', categories);
    legend('控制管', '同步管', 'Location', 'northwest');

%%-------------------------------------------------------------------------------------------------------------
    function [total_loss, breakdown] = calculate_mosfet_loss(varargin)
% 计算MOSFET功率损耗
% 用法1: calculate_mosfet_loss('param1', value1, 'param2', value2, ...)
% 用法2: [total_loss, breakdown] = calculate_mosfet_loss(config_struct)

    %% 参数解析
    if nargin == 1 && isstruct(varargin{1})
        config = varargin{1};
    else
        config = parse_parameters(varargin{:});
    end
    
    %% 从配置中提取参数
    Vin = config.Vin;           % 输入电压 (V)
    Vout = config.Vout;         % 输出电压 (V)
    I_load = config.I_load;     % 负载电流 (A)
    fsw = config.fsw;           % 开关频率 (Hz)
    Vgs = config.Vgs;           % 栅极驱动电压 (V)
    Tj_est = config.Tj_est;     % 估计结温 (°C)
    Ta = config.Ta;             % 环境温度 (°C)
    
    % MOSFET参数
    Rds_on = config.Rds_on;     % 导通电阻 (Ω)
    Qg = config.Qg;             % 总栅极电荷 (nC)
    Qgs = config.Qgs;           % 栅源电荷 (nC)
    Qgd = config.Qgd;           % 栅漏电荷 (nC)
    tr = config.tr;             % 上升时间 (s)
    tf = config.tf;             % 下降时间 (s)
    Coss = config.Coss;         % 输出电容 (F)
    Rth_ja = config.Rth_ja;     % 结到环境热阻 (°C/W)
    
    %% 计算基本工作参数
    % D = Vout / Vin;             % 占空比 (Buck电路)
    % Irms = I_load * sqrt(D);    % RMS电流 (简化计算)

    % 占空比计算
D = 1 - (Vin / Vout);
% RMS电流计算
Irms = I_load / (1 - D) * sqrt(1 - D); % 输入电流RMS
% 开关损耗电压
V_sw = Vout; % MOSFET关断时承受输出电压
    
    %% 1. 导通损耗计算
    % 考虑温度对Rds_on的影响 (典型: Rds_on随温度升高而增大)
    Rds_on_actual = Rds_on * (1 + 0.007 * (Tj_est - 25)); % 0.7%/°C的温度系数
    P_cond = Irms^2 * Rds_on_actual * D;
    
    %% 2. 开关损耗计算
    % 方法1: 使用开关时间计算
    E_switch_single = 0.5 * Vin * I_load * (tr + tf); % 单次开关能量 (J)
    P_sw_time = E_switch_single * fsw;
    
    % 方法2: 使用栅极电荷计算 (更精确)
    V_plateau = 4; % 典型米勒平台电压 (V)，可根据数据手册调整
    E_on = Qgd * 1e-9 * V_plateau;  % 开通能量 (J)
    E_off = Qgd * 1e-9 * (Vin - V_plateau); % 关断能量 (J)
    P_sw_charge = (E_on + E_off) * fsw;
    
    % 使用更精确的方法
    P_sw = P_sw_charge;
    
    %% 3. 输出电容损耗
    P_coss = 0.5 * Coss * Vin^2 * fsw;
    
    %% 4. 驱动损耗
    P_drive = Qg * 1e-9 * Vgs * fsw;
    
    %% 5. 总损耗计算
    P_total = P_cond + P_sw + P_coss + P_drive;
    
    %% 6. 热计算 - 迭代以获得更准确的结果
    Tj_new = Ta + P_total * Rth_ja;
    
    % 如果估计结温与实际计算相差较大，重新计算
    if abs(Tj_new - Tj_est) > 5
        fprintf('进行热迭代: Tj_est = %.1f°C, Tj_new = %.1f°C\n', Tj_est, Tj_new);
        % 更新Rds_on基于新的结温
        Rds_on_new = Rds_on * (1 + 0.007 * (Tj_new - 25));
        P_cond_new = Irms^2 * Rds_on_new * D;
        P_total_new = P_cond_new + P_sw + P_coss + P_drive;
        
        % 使用迭代后的值
        P_cond = P_cond_new;
        P_total = P_total_new;
        Tj_final = Ta + P_total_new * Rth_ja;
    else
        Tj_final = Tj_new;
    end
    
    %% 输出结果
    fprintf('\n=== MOSFET损耗分析结果 ===\n');
    fprintf('工作条件:\n');
    fprintf('  输入电压: %.1f V\n', Vin);
    fprintf('  输出电压: %.1f V\n', Vout);
    fprintf('  负载电流: %.1f A\n', I_load);
    fprintf('  开关频率: %.0f kHz\n', fsw/1e3);
    fprintf('  占空比: %.2f\n', D);
    
    fprintf('\n损耗分析:\n');
    fprintf('  导通损耗: %.3f W (%.1f%%)\n', P_cond, P_cond/P_total*100);
    fprintf('  开关损耗: %.3f W (%.1f%%)\n', P_sw, P_sw/P_total*100);
    fprintf('  Coss损耗: %.3f W (%.1f%%)\n', P_coss, P_coss/P_total*100);
    fprintf('  驱动损耗: %.3f W (%.1f%%)\n', P_drive, P_drive/P_total*100);
    fprintf('  -------------------------\n');
    fprintf('  总损耗:   %.3f W\n', P_total);
    
    fprintf('\n热分析:\n');
    fprintf('  环境温度: %.1f°C\n', Ta);
    fprintf('  结温:     %.1f°C\n', Tj_final);
    fprintf('  热裕量:   %.1f°C (假设最大结温150°C)\n', 150 - Tj_final);
    
    %% 返回结果
    total_loss = P_total;
    breakdown.cond_loss = P_cond;
    breakdown.sw_loss = P_sw;
    breakdown.coss_loss = P_coss;
    breakdown.drive_loss = P_drive;
    breakdown.junction_temp = Tj_final;
    
    %% 绘制损耗分布图
    if config.plot_results
        plot_loss_breakdown(breakdown);
    end
end

function config = parse_parameters(varargin)
% 解析输入参数
    p = inputParser;
    
    % 默认参数 - 基于典型的MOSFET应用
    addParameter(p, 'Vin', 24, @isnumeric);
    addParameter(p, 'Vout', 12, @isnumeric);
    addParameter(p, 'I_load', 5, @isnumeric);
    addParameter(p, 'fsw', 100e3, @isnumeric);
    addParameter(p, 'Vgs', 10, @isnumeric);
    addParameter(p, 'Tj_est', 80, @isnumeric);
    addParameter(p, 'Ta', 40, @isnumeric);
    
    % MOSFET参数默认值 (基于典型器件)
    addParameter(p, 'Rds_on', 0.01, @isnumeric);      % 10mΩ
    addParameter(p, 'Qg', 25, @isnumeric);            % 25nC
    addParameter(p, 'Qgs', 8, @isnumeric);            % 8nC
    addParameter(p, 'Qgd', 12, @isnumeric);           % 12nC
    addParameter(p, 'tr', 20e-9, @isnumeric);         % 20ns
    addParameter(p, 'tf', 15e-9, @isnumeric);         % 15ns
    addParameter(p, 'Coss', 300e-12, @isnumeric);     % 300pF
    addParameter(p, 'Rth_ja', 40, @isnumeric);        % 40°C/W
    
    addParameter(p, 'plot_results', true, @islogical);
    
    parse(p, varargin{:});
    config = p.Results;
end

function plot_loss_breakdown(breakdown)
% 绘制损耗分布饼图
    losses = [breakdown.cond_loss, breakdown.sw_loss, ...
              breakdown.coss_loss, breakdown.drive_loss];
    labels = {'导通损耗', '开关损耗', 'Coss损耗', '驱动损耗'};
    
    figure;
    pie(losses, labels);
    title('MOSFET损耗分布');
    
    % 添加数值标签
    legend_text = cell(1, length(labels));
    for i = 1:length(labels)
        legend_text{i} = sprintf('%s: %.3fW (%.1f%%)', ...
            labels{i}, losses(i), losses(i)/sum(losses)*100);
    end
    legend(legend_text, 'Location', 'eastoutside');
end
